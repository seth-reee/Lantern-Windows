cmake_minimum_required(VERSION 3.24)
project(Lantern VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Concurrent)
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0004 NEW)
if(MSVC)
 add_compile_options(/utf-8 /Zc:__cplusplus)
endif()
add_library(scanner src/scanner.cpp src/scanner.h src/networkprobe.cpp src/networkprobe.h src/diagnosticlog.cpp src/diagnosticlog.h)
target_include_directories(scanner PUBLIC src)
target_link_libraries(scanner PUBLIC Qt6::Network Qt6::Gui Qt6::Concurrent)
if(WIN32)
 target_link_libraries(scanner PRIVATE iphlpapi ws2_32)
 target_compile_definitions(scanner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
endif()
qt_add_executable(lantern src/main.cpp src/theme.h)
target_compile_definitions(lantern PRIVATE LANTERN_VERSION="${PROJECT_VERSION}")
qt_add_resources(lantern about_assets PREFIX "/" FILES Lantern.png resources/icons/lantern-black.png resources/icons/lantern-white.png)
qt_add_qml_module(lantern URI Lantern VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(lantern PRIVATE scanner Qt6::Quick Qt6::QuickControls2)
if(WIN32)
 set_target_properties(lantern PROPERTIES WIN32_EXECUTABLE TRUE)
 configure_file(packaging/windows/lantern.rc.in "${CMAKE_CURRENT_BINARY_DIR}/lantern.rc" @ONLY)
 target_sources(lantern PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/lantern.rc")
 target_include_directories(lantern PRIVATE packaging/windows)
 install(TARGETS lantern RUNTIME DESTINATION .)
 install(FILES LICENSE packaging/windows/THIRD-PARTY-NOTICES.txt DESTINATION .)
else()
 install(TARGETS lantern RUNTIME DESTINATION bin)
 install(FILES packaging/lantern.desktop DESTINATION share/applications)
 install(FILES Lantern.png DESTINATION share/pixmaps RENAME lantern.png)
 install(FILES LICENSE DESTINATION share/licenses/lantern)
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(scanner_tests tests/scanner_test.cpp)
 target_link_libraries(scanner_tests PRIVATE scanner Qt6::Test)
 add_test(NAME scanner COMMAND scanner_tests)
 qt_add_executable(ui_tests tests/ui_test.cpp src/theme.h)
 qt_add_resources(ui_tests about_assets PREFIX "/" FILES Lantern.png resources/icons/lantern-black.png resources/icons/lantern-white.png)
 target_link_libraries(ui_tests PRIVATE scanner Qt6::Test Qt6::Quick Qt6::QuickControls2)
 target_compile_definitions(ui_tests PRIVATE LANTERN_QML="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
 add_test(NAME ui_results COMMAND ui_tests)
 set_tests_properties(ui_results PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=basic;QT_QUICK_BACKEND=software")
 add_test(NAME ui_smoke COMMAND lantern --smoke-test)
 set_tests_properties(scanner ui_results ui_smoke PROPERTIES TIMEOUT 60)
 set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=basic;QT_QUICK_BACKEND=software")
endif()
